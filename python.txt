class GaugeWidget(QWidget):
    def __init__(self, title, unit, color, parent=None):
        super().__init__(parent)
        self.title = title
        self.unit = unit
        self.color = QColor(color)
        self.value = 0
        self.sub = ""
        self.setMinimumHeight(160)
        self.setSizePolicy(QSizePolicy.Expanding, QSizePolicy.Fixed)

    def set_value(self, value, sub=""):
        self.value = int(value)
        self.sub = sub
        self.update()

    def paintEvent(self, e):
        p = QPainter(self)
        p.setRenderHint(QPainter.Antialiasing)

        rect = self.rect().adjusted(12, 12, -12, -12)
        center = rect.center()
        radius = min(rect.width(), rect.height()) // 2 - 14

        p.setPen(Qt.NoPen)
        p.setBrush(QColor(12, 18, 24))
        p.drawRoundedRect(self.rect().adjusted(8, 8, -8, -8), 16, 16)

        bg_pen = QPen(QColor(40, 55, 68), 12)
        bg_pen.setCapStyle(Qt.RoundCap)
        p.setPen(bg_pen)
        p.drawArc(center.x() - radius, center.y() - radius, radius * 2, radius * 2, 45 * 16, -360 * 16)

        span = int(360 * 16 * clamp(self.value / 100.0, 0.0, 1.0))
        fg_pen = QPen(self.color, 12)
        fg_pen.setCapStyle(Qt.RoundCap)
        p.setPen(fg_pen)
        p.drawArc(center.x() - radius, center.y() - radius, radius * 2, radius * 2, 225 * 16, -span)

        p.setPen(QColor("#e8fbff"))
        f = QFont("Segoe UI", 10, QFont.DemiBold)
        p.setFont(f)
        p.drawText(QRect(0, 14, self.width(), 24), Qt.AlignHCenter | Qt.AlignTop, self.title)

        f2 = QFont("Segoe UI", 24, QFont.Bold)
        p.setFont(f2)
        p.setPen(self.color)
        p.drawText(QRect(0, self.height()//2 - 20, self.width(), 44), Qt.AlignHCenter | Qt.AlignVCenter, f"{self.value}")

        f3 = QFont("Segoe UI", 10)
        p.setFont(f3)
        p.setPen(QColor("#97b8c7"))
        p.drawText(QRect(0, self.height()//2 + 18, self.width(), 26), Qt.AlignHCenter | Qt.AlignTop, self.unit + (f" · {self.sub}" if self.sub else ""))

class ReaktorWidget(QWidget):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.setFixedSize(110, 110)
        self.pulse = 0.0
        self.pix = QPixmap()
        self.timer = QTimer(self)
        self.timer.timeout.connect(self._animate)
        self.timer.start(30)
        self.load_default()

    def load_default(self):
        self.pix = QPixmap(110, 110)
        self.pix.fill(Qt.transparent)
        p = QPainter(self.pix)
        p.setRenderHint(QPainter.Antialiasing)
        path = QPainterPath()
        path.moveTo(55, 10)
        path.lineTo(95, 90)
        path.lineTo(15, 90)
        path.closeSubpath()
        grad = QLinearGradient(20, 20, 90, 95)
        grad.setColorAt(0, QColor("#b5ffff"))
        grad.setColorAt(0.5, QColor("#00f0ff"))
        grad.setColorAt(1, QColor("#0a3a55"))
        p.setPen(QPen(QColor("#66f8ff"), 4))
        p.setBrush(QBrush(grad))
        p.drawPath(path)
        p.setBrush(Qt.NoBrush)
        p.setPen(QPen(QColor("#ff4b4b"), 2))
        p.drawEllipse(40, 40, 30, 30)
        p.end()

    def _animate(self):
        self.pulse += 0.06
        self.update()

    def paintEvent(self, e):
        p = QPainter(self)
        p.setRenderHint(QPainter.Antialiasing)
        glow = 70 + int(35 * (0.5 + 0.5 * __import__("math").sin(self.pulse)))
        p.fillRect(self.rect(), Qt.transparent)
        p.setBrush(QColor(8, 12, 18, 220))
        p.setPen(QPen(QColor(0, 240, 255, 90), 1))
        p.drawRoundedRect(self.rect().adjusted(4, 4, -4, -4), 18, 18)

        shadow = QColor(0, 240, 255, glow)
        p.setPen(QPen(shadow, 10))
        p.drawRoundedRect(self.rect().adjusted(8, 8, -8, -8), 16, 16)

        if not self.pix.isNull():
            img = self.pix.scaled(self.size(), Qt.KeepAspectRatioByExpanding, Qt.SmoothTransformation)
            p.setClipPath(QPainterPath())
            p.drawPixmap(0, 0, img)

    def set_image(self, img: QImage):
        self.pix = QPixmap.fromImage(img).scaled(self.size(), Qt.KeepAspectRatioByExpanding, Qt.SmoothTransformation)
        self.update()

class CameraWorker(QObject):
    frameReady = Signal(QImage)
    error = Signal(str)

    def __init__(self):
        super().__init__()
        self.capture_once = False
        self.running = True

    def request_capture(self):
        self.capture_once = True

    def stop(self):
        self.running = False

    def run(self):
        if cv2 is None:
            self.error.emit("OpenCV unavailable")
            return

        cap = None
        try:
            cap = cv2.VideoCapture(0, cv2.CAP_DSHOW)
            if not cap.isOpened():
                cap = cv2.VideoCapture(0)
            if not cap.isOpened():
                self.error.emit("Camera open failed")
                return

            cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)
            last = time.time()
            while self.running:
                ok, frame = cap.read()
                if ok:
                    frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                    h, w, ch = frame.shape
                    img = QImage(frame.data, w, h, ch * w, QImage.Format_RGB888).copy()
                    if self.capture_once:
                        self.frameReady.emit(img)
                        self.capture_once = False
                else:
                    if time.time() - last > 3:
                        self.error.emit("Camera frame read failed")
                        last = time.time()
                time.sleep(0.03)
        except Exception as ex:
            self.error.emit(str(ex))
        finally:
            if cap is not None:
                cap.release()

class OpenRouterWorker(QObject):
    responseReady = Signal(str)
    error = Signal(str)

    def __init__(self):
        super().__init__()
        self._q = queue.Queue()
        self._running = True

    def ask(self, text, history):
        self._q.put((text, history))

    def stop(self):
        self._running = False
        self._q.put(None)

    def run(self):
        if requests is None:
            self.error.emit("requests unavailable")
            return
        sess = requests.Session()
        sess.trust_env = False
        headers = {
            "Authorization": f"Bearer {OPENROUTER_KEY}",
            "Content-Type": "application/json",
            "HTTP-Referer": "http://localhost",
            "X-Title": "JARVIS HUD",
        }
        while self._running:
            item = self._q.get()
            if item is None:
                continue
            text, history = item
            try:
                messages = [{"role": "system", "content": SYSTEM_PROMPT + "\n" + MEMORY_PROMPT}]
                for role, content in history[-12:]:
                    messages.append({"role": role, "content": content})
                messages.append({"role": "user", "content": text})

                payload = {
                    "model": MODEL_NAME,
                    "messages": messages,
                    "temperature": 0.4,
                }
                resp = sess.post(
                    OPENROUTER_URL,
                    headers=headers,
                    data=json.dumps(payload),
                    timeout=None,
                )
                resp.raise_for_status()
                data = resp.json()
                content = data["choices"][0]["message"]["content"]
                self.responseReady.emit(content.strip())
            except Exception as ex:
                self.error.emit(str(ex))

    def _build_timers(self):
        self.system_timer = QTimer(self)
        self.system_timer.timeout.connect(self.refresh_system)
        self.system_timer.start(1000)

    def refresh_system(self):
        cpu = int(psutil.cpu_percent(interval=None))
        gpu_temp = self.get_gpu_temp()
        if gpu_temp is None:
            gpu_temp = 0
        self.cpu_gauge.set_value(cpu, "dynamic")
        self.gpu_gauge.set_value(clamp(gpu_temp, 0, 100), "sensor")

    def get_gpu_temp(self):
        try:
            out = subprocess.check_output(
                ["nvidia-smi", "--query-gpu=temperature.gpu", "--format=csv,noheader,nounits"],
                stderr=subprocess.DEVNULL,
                text=True,
                timeout=2,
                creationflags=subprocess.CREATE_NO_WINDOW if hasattr(subprocess, "CREATE_NO_WINDOW") else 0,
            )
            vals = [int(x.strip()) for x in out.splitlines() if x.strip().isdigit()]
            return vals[0] if vals else None
        except Exception:
            return None

    def capture_frame(self):
        self.captureSignal.emit()

    def on_camera_frame(self, img):
        scaled = img.scaled(self.camera_preview.size(), Qt.KeepAspectRatioByExpanding, Qt.SmoothTransformation)
        self.camera_preview.setPixmap(QPixmap.fromImage(scaled))

    def on_worker_error(self, msg):
        self.append_chat_system(f"ERR: {msg}")

    def append_chat_user(self, text):
        self.chat_history.append(f"<span style='color:#7befff'><b>YOU:</b></span> {self.escape(text)}")
        self.scroll_chat_bottom()

    def append_chat_assistant(self, text):
        self.chat_history.append(f"<span style='color:#ffb8b8'><b>JARVIS:</b></span> {self.escape(text)}")
        self.scroll_chat_bottom()

    def append_chat_system(self, text):
        self.chat_history.append(f"<span style='color:#ffcc66'><b>SYSTEM:</b></span> {self.escape(text)}")
        self.scroll_chat_bottom()

    def escape(self, text):
        return (
            text.replace("&", "&amp;")
            .replace("<", "&lt;")
            .replace(">", "&gt;")
            .replace("\n", "<br>")
        )

    def scroll_chat_bottom(self):
        sb = self.chat_history.verticalScrollBar()
        sb.setValue(sb.maximum())

    def send_chat(self):
        text = self.chat_input.text().strip()
        if not text:
            return
        self.chat_input.clear()
        self.append_chat_user(text)
        self.history.append(("user", text))
        if not OPENROUTER_KEY:
            self.append_chat_system("OpenRouter key missing.")
            return
        self.askSignal.emit(text)

    def on_ai_response(self, text):
        self.history.append(("assistant", text))
        self.append_chat_assistant(text)

    def closeEvent(self, e):
        try:
            self.cam_worker.stop()
            self.ai_worker.stop()
            self.cam_thread.quit()
            self.cam_thread.wait(1500)
            self.ai_thread.quit()
            self.ai_thread.wait(1500)
        except Exception:
            pass
        e.accept()
